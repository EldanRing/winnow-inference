cmake_minimum_required(VERSION 3.24)
project(winnow LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_COMMON ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_TOOLS ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_SERVER ON CACHE BOOL "" FORCE)
set(LLAMA_BUILD_MTMD ON CACHE BOOL "" FORCE)
set(MTMD_VIDEO OFF CACHE BOOL "" FORCE)
set(LLAMA_BUILD_UI OFF CACHE BOOL "" FORCE)
set(LLAMA_USE_PREBUILT_UI OFF CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory(.runtime/llama.cpp runtime EXCLUDE_FROM_ALL)
target_sources(server-context PRIVATE native/bridge.cpp)
target_include_directories(server-context PUBLIC "${CMAKE_SOURCE_DIR}/native")
target_include_directories(server-context PRIVATE "${CMAKE_SOURCE_DIR}/.runtime/llama.cpp/vendor")
file(SHA256 "${CMAKE_SOURCE_DIR}/runtime.lock.json" WINNOW_RUNTIME_SHA256)
target_compile_definitions(server-context PRIVATE WINNOW_RUNTIME_SHA256="${WINNOW_RUNTIME_SHA256}")
if(APPLE)
  find_library(ACCELERATE_FRAMEWORK Accelerate REQUIRED)
  target_link_libraries(server-context PRIVATE "${ACCELERATE_FRAMEWORK}")
  target_compile_definitions(server-context PRIVATE ACCELERATE_NEW_LAPACK)
endif()
set_target_properties(llama-server PROPERTIES OUTPUT_NAME winnow-server RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
enable_testing()
add_executable(winnow-unit tests/unit.cpp)
target_include_directories(winnow-unit PRIVATE native .runtime/llama.cpp/vendor)
target_compile_options(winnow-unit PRIVATE -UNDEBUG)
add_test(NAME winnow-unit COMMAND winnow-unit)
